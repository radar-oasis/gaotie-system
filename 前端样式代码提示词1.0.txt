import React, { useState, useRef, useEffect } from 'react';
import { Button, Tag, Avatar, Textarea, MessagePlugin } from 'tdesign-react';
import {
  RefreshIcon,
  FileCopyIcon,
  ThumbUpIcon,
  ThumbDownIcon,
  ShareIcon,
  SendIcon,
  UserIcon,
  LogoCodepenIcon // 使用一个图标代指 AI 头像
} from 'tdesign-icons-react';
import 'tdesign-react/dist/tdesign.css'; // 引入 TDesign 全局样式

// --- 类型定义 ---
interface Message {
  id: string;
  role: 'user' | 'ai';
  content: string;
}

interface SelectedTag {
  id: string;
  label: string;
  topic?: string;
}

// --- 核心组件 ---
export default function ChatWorkspace() {
  const [inputValue, setInputValue] = useState('');
  const [messages, setMessages] = useState<Message[]>([
    {
      id: '1',
      role: 'ai',
      content: '我已经提取了硬性约束。关于设计主题，针对“中国文化”，我们可以联想到京张铁路沿线的“边塞文化”；针对“奥运元素”，我们可以聚焦“冰雪运动”。\n\n你倾向于哪个方向的具体意象？\n\n[TAG: 中国文化-龙]\n[TAG: 中国文化-长城]\n[TAG: 冰雪运动-雪花]\n[TAG: 冰雪运动-滑冰]'
    }
  ]);
  const [selectedTags, setSelectedTags] = useState<SelectedTag[]>([]);
  const chatEndRef = useRef<HTMLDivElement>(null);

  // 自动滚动到底部
  useEffect(() => {
    chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  // 处理发送消息
  const handleSend = () => {
    if (!inputValue.trim()) return;
    
    const newUserMsg: Message = { id: Date.now().toString(), role: 'user', content: inputValue };
    setMessages((prev) => [...prev, newUserMsg]);
    setInputValue('');

    // 模拟 AI 回复 (真实场景需对接大模型接口)
    setTimeout(() => {
      const newAiMsg: Message = {
        id: (Date.now() + 1).toString(),
        role: 'ai',
        content: `好的，已收到您的想法。结合您的需求，我们可以进一步发散这些意象：\n\n[TAG: 燕山山脉]\n[TAG: 百年京张轨道线]`
      };
      setMessages((prev) => [...prev, newAiMsg]);
    }, 1000);
  };

  // 处理点击标签加入右侧清单
  const handleTagClick = (tagStr: string) => {
    const isExist = selectedTags.some(t => t.label === tagStr);
    if (!isExist) {
      let topic = undefined;
      let label = tagStr;
      if (tagStr.includes('-')) {
        const parts = tagStr.split('-');
        topic = parts[0];
        label = parts[1];
      }
      setSelectedTags(prev => [...prev, { id: Date.now().toString(), label, topic }]);
      MessagePlugin.success(`已添加意象：${label}`);
    }
  };

  // 渲染 AI 消息内容（解析 [TAG: xxx]）
  const renderAiContent = (content: string) => {
    // 按 [TAG: xxx] 切割字符串
    const parts = content.split(/(\[TAG:.*?\])/g);
    
    return parts.map((part, index) => {
      const match = part.match(/\[TAG:\s*(.+?)\]/);
      if (match) {
        const tagStr = match[1];
        const isSelected = selectedTags.some(t => t.label === tagStr || t.label === tagStr.split('-')[1]);
        
        return (
          <Tag 
            key={index} 
            theme={isSelected ? "primary" : "default"} 
            variant="outline" 
            style={{ margin: '4px 8px 4px 0', cursor: isSelected ? 'default' : 'pointer' }}
            onClick={() => !isSelected && handleTagClick(tagStr)}
          >
            {isSelected ? `✓ ${tagStr}` : `+ ${tagStr}`}
          </Tag>
        );
      }
      // 普通文本保留换行
      return <span key={index} style={{ whiteSpace: 'pre-wrap' }}>{part}</span>;
    });
  };

  // 渲染 AI 底部操作栏 (复刻截图)
  const renderActionBar = () => (
    <div style={{ display: 'flex', gap: '4px', marginTop: '8px', background: '#f3f4f6', padding: '4px 8px', borderRadius: '4px', width: 'fit-content' }}>
      <Button variant="text" shape="square" size="small"><RefreshIcon /></Button>
      <Button variant="text" shape="square" size="small"><FileCopyIcon /></Button>
      <Button variant="text" shape="square" size="small"><ThumbUpIcon /></Button>
      <Button variant="text" shape="square" size="small"><ThumbDownIcon /></Button>
      <Button variant="text" shape="square" size="small"><ShareIcon /></Button>
    </div>
  );

  return (
    <div style={{ display: 'flex', height: '100vh', width: '100%', backgroundColor: '#fcfcfc', fontFamily: 'system-ui, sans-serif' }}>
      
      {/* 左侧：对话区域 */}
      <div style={{ flex: '1', display: 'flex', flexDirection: 'column', borderRight: '1px solid #e7e7e7' }}>
        
        {/* 对话列表 */}
        <div style={{ flex: '1', overflowY: 'auto', padding: '24px' }}>
          {messages.map((msg) => (
            <div 
              key={msg.id} 
              style={{ 
                display: 'flex', 
                marginBottom: '24px', 
                flexDirection: msg.role === 'user' ? 'row-reverse' : 'row',
                gap: '16px'
              }}
            >
              {/* 头像 */}
              <Avatar shape="circle" style={{ backgroundColor: msg.role === 'user' ? '#e2e8f0' : '#0052d9', color: msg.role === 'user' ? '#475569' : '#fff', flexShrink: 0 }}>
                {msg.role === 'user' ? <UserIcon /> : <LogoCodepenIcon />}
              </Avatar>

              {/* 消息体 */}
              <div style={{ 
                display: 'flex', 
                flexDirection: 'column', 
                alignItems: msg.role === 'user' ? 'flex-end' : 'flex-start',
                maxWidth: '75%' 
              }}>
                <div style={{ 
                  backgroundColor: msg.role === 'user' ? '#f3f4f6' : '#ffffff', 
                  border: msg.role === 'ai' ? '1px solid #e7e7e7' : 'none',
                  padding: '12px 16px', 
                  borderRadius: '8px',
                  color: '#333',
                  lineHeight: '1.6',
                  fontSize: '14px',
                  boxShadow: msg.role === 'ai' ? '0 2px 8px rgba(0,0,0,0.02)' : 'none'
                }}>
                  {msg.role === 'user' ? msg.content : renderAiContent(msg.content)}
                </div>
                
                {/* AI 专属操作栏 */}
                {msg.role === 'ai' && renderActionBar()}
              </div>
            </div>
          ))}
          <div ref={chatEndRef} />
        </div>

        {/* 底部输入框区域 */}
        <div style={{ padding: '24px', backgroundColor: '#fff', borderTop: '1px solid #e7e7e7' }}>
          <div style={{ position: 'relative', border: '1px solid #dcdcdc', borderRadius: '8px', overflow: 'hidden', paddingBottom: '40px' }}>
             <Textarea
                placeholder="请输入消息..."
                value={inputValue}
                onChange={(v) => setInputValue(v)}
                onEnter={handleSend}
                autosize={{ minRows: 2, maxRows: 5 }}
                style={{ border: 'none', boxShadow: 'none' }}
              />
              <Button 
                shape="circle" 
                theme={inputValue.trim() ? "primary" : "default"}
                style={{ position: 'absolute', right: '12px', bottom: '8px' }}
                onClick={handleSend}
                disabled={!inputValue.trim()}
              >
                <SendIcon />
              </Button>
          </div>
        </div>
      </div>

      {/* 右侧：意象收集箱 (保留原有业务逻辑) */}
      <div style={{ width: '320px', backgroundColor: '#f9fafb', display: 'flex', flexDirection: 'column' }}>
        <div style={{ padding: '20px', borderBottom: '1px solid #e7e7e7', fontWeight: 'bold' }}>
          当前设计目标清单
        </div>
        <div style={{ padding: '20px', flex: 1, overflowY: 'auto' }}>
           {selectedTags.length === 0 ? (
             <div style={{ color: '#999', fontSize: '14px', textAlign: 'center', marginTop: '40px' }}>暂无意象，请在对话中点击提取</div>
           ) : (
             <div style={{ display: 'flex', flexDirection: 'column', gap: '8px' }}>
                {selectedTags.map(tag => (
                  <div key={tag.id} style={{ display: 'flex', justifyContent: 'space-between', padding: '8px 12px', backgroundColor: '#fff', border: '1px solid #e7e7e7', borderRadius: '4px' }}>
                     <span style={{ fontSize: '14px' }}>
                        {tag.topic && <span style={{ color: '#888', marginRight: '4px' }}>{tag.topic}-</span>}
                        {tag.label}
                     </span>
                     <span style={{ color: '#d54941', cursor: 'pointer' }} onClick={() => setSelectedTags(prev => prev.filter(t => t.id !== tag.id))}>×</span>
                  </div>
                ))}
             </div>
           )}
        </div>
        <div style={{ padding: '20px' }}>
          <Button block theme="primary" size="large" disabled={selectedTags.length === 0}>
            确认目标，生成设计
          </Button>
        </div>
      </div>

    </div>
  );
}
